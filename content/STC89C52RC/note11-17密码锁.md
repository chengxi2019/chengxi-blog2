需要delay,display,key函数配合使用 b 
一开始先显示`------`,输入密码后数组向左移，正确后显示`-----P`,错误显示`-----E`  
目前缺失key.c,key.h文件收录  

``` c
#include <reg52.h>
#include "delay.h"
#include "display.h"
#include "key.h"

sbit LED = P2^0;
unsigned char code passwd[6] = {1,1,1,1,1,1};
unsigned char input[6];
unsigned char correct;
void key_lock();
void main()
{
    LED = 1;
    while(1)
    {
        key_lock();
    }
}

void key_lock()
{
    unsigned char count;
    unsigned char i,j,key;

    for(i=2;i<8;i++)
    {
        smg_buf[i]=16;
    }
    count = 0;

    while(count<6)
    {
        smg_display();
        key = key_scan();
        if(key != 0xff)
        {
            for(i=2;i<7;i++)
            {
                smg_buf[i] = smg_buf[i+1];
            }
            smg_buf[7]=key;
            input[count]=key;
            count++;

            while(key_scan() != 0xff)
            {
                smg_display();
            }
        }
    }

    correct = 1;
    for(i=0;i<6;i++)
    {
        if(input[i]!=passwd[i])
        {
            correct = 0;
            break;
        }
    }

    for(i=2;i<7;i++)
    {
        smg_buf[i]=16;
    }
    if(correct==1)
    {
        smg_buf[7]=17;
        LED = 0;
        for(j=0;j<150;j++)
        {
            smg_display();
        }
    }
    else
    {
        smg_buf[7] = 14;
        LED =1;
        for(j=0;j<150;j++)
        {
            smg_display();
        }
    }
}
```
