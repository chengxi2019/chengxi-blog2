# 普中科技STC89C52RC文件分发处
适用于普中科技STC89C52RC单片机的文件  

### 延时函数delay.c

``` c
#include "delay.h"
void delay_10us(unsigned int ten_us)
{
	while(ten_us--);	
}

void delay(unsigned int t)
{
	unsigned int i,j;
	for(i=t;i>0;i--)
	for(j=100;j>0;j--);
}
```

### 数码管显示函数display.c

#include <reg52.h>

#include "display.h"
unsigned char code smg_tab[]={
			0x3f,  //0

			0x6d,  //5

			0x39,  //C
			0x5e,  //D
			0x79,  //E
			0x71,  //F
			0x40,  //-
	    	0x00,  //熄灭
};
//数据缓冲区
unsigned char smg_buf[]={17,17,17,17,16,16,16,16};
void smg_display(void)
{
	static unsigned char i=0;
	   	switch(i)//位选
		{
			case 0: 
				SMG_PORT = smg_tab[smg_buf[0]]; //送段码
				LSC=1;LSB=1;LSA=1;  //送位选
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;
				break;  //千万位
			
			case 1: 	
				SMG_PORT = smg_tab[smg_buf[1]]; //送段码
				LSC=1;LSB=1;LSA=0;  //送位选
				delay_10us(100);  //延时
				i++;	
				break;  //百万位
	
			case 2: 
				SMG_PORT=0x00;//消隐
				SMG_PORT = smg_tab[smg_buf[2]]; //送段码
				LSC=1;LSB=0;LSA=1;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;	
		   		break;  //十万位
			
			case 3: 
				SMG_PORT = smg_tab[smg_buf[3]]; //送段码
				LSC=1;LSB=0;LSA=0;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;	
				break;  //万位
			
			case 4:
				SMG_PORT = smg_tab[smg_buf[4]]; //送段码
				LSC=0;LSB=1;LSA=1;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;	
			  break;  //千位
			case 5: 
				SMG_PORT = smg_tab[smg_buf[5]]; //送段码
				LSC=0;LSB=1;LSA=0;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;	
		    break;  //百位
			case 6: 
				SMG_PORT = smg_tab[smg_buf[6]]; //送段码
				LSC=0;LSB=0;LSA=1;
				delay_10us(100);  //延时
			  SMG_PORT=0x00;//消隐
				i++;	
			  break;  //十位
			case 7: 
				SMG_PORT = smg_tab[smg_buf[7]]; //送段码
				LSC=0;LSB=0;LSA=0;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i=0;	
			  break;  //个位
				default:break;
		}
}

### 按键读取函数key_scan.c
``` c
#include "key_scan.h"
#include "delay.h"
/*******按键扫描底层逻辑函数高阶*******/
unsigned char key_scan()
{
  unsigned char key_num=0;
  if(!key1 || !key0 || !key2 || !key3)   //按键没有按下去，目前状态都是处于假的状态	，有1个按下，表示状态为真
  {
  	delay(10);
	if(!key1 || !key0 || !key2 || !key3)
	{
		if(!key1) key_num = 1;
		if(!key0) key_num = 2; 
		if(!key2) key_num = 3;
		if(!key3) key_num = 4;
	}
  }	while(!key1 || !key0 || !key3 || !key3);
 return key_num;
}
```

### LCD屏显示低层函数lcd1602.c  
``` c
#include "lcd1602.h"
#include "delay.h"	  
#include <intrins.h>
/********读忙的操作*********/
void lcd_busy() //读液晶状态
{   unsigned char flag;             //判断标志
     while(1)
    {
		RS=0;
		RW=1;       // RW=01，RS=0
	    EN=1;
        _nop_();
		_nop_();
		_nop_();
		_nop_();  //延时
		flag=GPIO;    //读取端口的值，判断DB7状态
		EN=0;	   //将使能端置0以完成高脉冲向低脉冲跳变
		flag=flag & 0x80;  //保留最高位，判断DB7为1，忙，为0则不忙
		if(flag==0x00) break;  //不忙则跳出
     }//否则死循环再次判断
}  
/********写命令的操作*********/
void xie_ml(unsigned char com) //写命令
{
    lcd_busy();   //读取液晶状态是否忙
	RS=0;
	RW=0;       // RW=0，RS=0，写LCD命令字
	GPIO=com;    //将com中的命令字写入LCD数据口
	EN=1;	   //使能端给一高脉冲，
	delay(5);    //稍做延时
	EN=0;      //将使能端置0以完成由高脉冲向低脉冲跳变
} 
/********写数据的操作*********/
void xie_shuju(unsigned char shuju) //写数据
{
	lcd_busy();   //读取液晶状态是否忙
	RS=1;
	RW=0;       // RW=0，RS=1，写数据
	GPIO=shuju;    //将com中的命令字写入LCD数据口
	EN=1;	   //使能端给一高脉冲
	delay(5);    //稍做延时
	EN=0;      //将使能端置0以完成高脉冲向低脉冲跳变
}  
/********液晶的初始化********/
void LCD_csh() //液晶初始化子函数
{
	xie_ml(0x38);  //设置液晶2行显示，5×7点阵，8位数据显示
	xie_ml(0x0c);  //设置开始显示 不显示光标
	xie_ml(0x06);  //设置液晶读或写一个字符后，指针加1，向右移动	
    xie_ml(0x01);  //清屏
} 
/********写一个字符的函数底层逻辑*******/
void  lcd_showchar(unsigned char lie,unsigned char hang,unsigned char dat)
{       
	if(hang==0)          //第一行
	{
		xie_ml(0x80+lie);//发送第一行地址命令
	}
	else	
	{
		xie_ml(0xc0+lie);//发送第二行地址命令
	}
	xie_shuju(dat);  //发送数据
}
```
