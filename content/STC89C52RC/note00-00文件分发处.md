# 普中科技STC89C52RC文件分发处
适用于普中科技STC89C52RC单片机的文件  

### 延时函数delay.c

``` c
#include "delay.h"
void delay_10us(unsigned int ten_us)
{
	while(ten_us--);	
}

void delay(unsigned int t)
{
	unsigned int i,j;
	for(i=t;i>0;i--)
	for(j=100;j>0;j--);
}
```

### 延时函数delay.h(头文件)
``` c
#ifndef __DELAY_H__
#define __DELAY_H__

void delay_10us(unsigned int ten_us);
void delay(unsigned int t);
#endif
```

### 数码管显示函数display.c

``` c
#include <reg52.h>
#include "delay.h"
#include "display.h"
unsigned char code smg_tab[]={
			0x3f,  //0
			0x06,  //1
			0x5b,  //2
			0x4f,  //3
			0x66,  //4
			0x6d,  //5
			0x7d,  //6
			0x07,  //7
			0x7f,  //8
			0x6f,  //9
			0x77,  //A
			0x7c,  //B
			0x39,  //C
			0x5e,  //D
			0x79,  //E
			0x71,  //F
			0x40,  //-
	    	0x00,  //熄灭
};
//数据缓冲区
unsigned char smg_buf[]={17,17,17,17,16,16,16,16};
void smg_display(void)
{
	static unsigned char i=0;
	   	switch(i)//位选
		{
			case 0: 
				SMG_PORT = smg_tab[smg_buf[0]]; //送段码
				LSC=1;LSB=1;LSA=1;  //送位选
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;
				break;  //千万位
			
			case 1: 	
				SMG_PORT = smg_tab[smg_buf[1]]; //送段码
				LSC=1;LSB=1;LSA=0;  //送位选
				delay_10us(100);  //延时
				i++;	
				break;  //百万位
	
			case 2: 
				SMG_PORT=0x00;//消隐
				SMG_PORT = smg_tab[smg_buf[2]]; //送段码
				LSC=1;LSB=0;LSA=1;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;	
		   		break;  //十万位
			
			case 3: 
				SMG_PORT = smg_tab[smg_buf[3]]; //送段码
				LSC=1;LSB=0;LSA=0;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;	
				break;  //万位
			
			case 4:
				SMG_PORT = smg_tab[smg_buf[4]]; //送段码
				LSC=0;LSB=1;LSA=1;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;	
			  break;  //千位
			case 5: 
				SMG_PORT = smg_tab[smg_buf[5]]; //送段码
				LSC=0;LSB=1;LSA=0;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i++;	
		    break;  //百位
			case 6: 
				SMG_PORT = smg_tab[smg_buf[6]]; //送段码
				LSC=0;LSB=0;LSA=1;
				delay_10us(100);  //延时
			  SMG_PORT=0x00;//消隐
				i++;	
			  break;  //十位
			case 7: 
				SMG_PORT = smg_tab[smg_buf[7]]; //送段码
				LSC=0;LSB=0;LSA=0;
				delay_10us(100);  //延时
				SMG_PORT=0x00;//消隐
				i=0;	
			  break;  //个位
				default:break;
		}
}
```

### 按键读取函数key_scan.c
``` c
#include "key_scan.h"
#include "delay.h"
/*******按键扫描底层逻辑函数高阶*******/
unsigned char key_scan()
{
  unsigned char key_num=0;
  if(!key1 || !key0 || !key2 || !key3)   //按键没有按下去，目前状态都是处于假的状态	，有1个按下，表示状态为真
  {
  	delay(10);
	if(!key1 || !key0 || !key2 || !key3)
	{
		if(!key1) key_num = 1;
		if(!key0) key_num = 2; 
		if(!key2) key_num = 3;
		if(!key3) key_num = 4;
	}
  }	while(!key1 || !key0 || !key3 || !key3);
 return key_num;
}
```
