包含多选框，单选框，滑块，评分  

``` ts
import json from '@ohos.util.json';

@Entry
@Component
struct Index {
  @State gender:string = '';
  @State vData:number = 30;
  @State rNum:number = 4;

  //Row行容器
  //通用分离容器方式：容器间距：margin({方向：距离})
  build() {
     Column(){

     //多选框示例：
        Text("名著阅读调查").fontSize(36).fontWeight(600)
        Row(){
         CheckboxGroup({group:"books"})
           //附加部分:onchange:
           .onChange((item:CheckboxGroupResult)=>{console.log("已阅读书籍："+json.stringify(item))})
         Text("全选").fontSize(24).fontWeight(600)
        }.width('100%').margin({left:30})
     //多选框示例：
        Row(){
          Checkbox({name:"sanguo",group:"books"})
          Text("三国演义").fontSize(22)
        }.width('100%').margin({left:60})
        Row(){
          Checkbox({name:"shuihu",group:"books"}).shape(CheckBoxShape.ROUNDED_SQUARE)//更改多选框形状
          Text("水浒传").fontSize(22)
        }.width('100%').margin({left:60})
        Row(){
          Checkbox({name:"xiyou",group:"books"})
          Text("西游记").fontSize(22)
        }.width('100%').margin({left:60})
        Row(){
          Checkbox({name:"honlou",group:"books"})
          Text("红楼梦").fontSize(22)
        }.width('100%').margin({left:60})

     //单选框示例：
        Text()
        Row(){
        Radio({value:"male",group:"sexy"}).onChange((isChecked:boolean)=>{if(isChecked) this.gender='男'});
        Text("男").fontSize(22).margin({right:30})
    
        Radio({value:"female",group:"sexy"}).checked(true).onChange((isChecked:boolean)=>{if(isChecked) this.gender='女'});
        Text("女").fontSize(22)      
        }
        Text(`你是:${this.gender}生`).fontSize(22)
 
     //滑块:
        Slider({value:30,min:20,max:60}).showTips(true)
          .onChange((value: number, mode: SliderChangeMode) => {
            this.vData = value;
          })     
        Text("当前音量："+this.vData).fontSize(20)
 
    //评分：
       Rating({rating:this.rNum}).stars(5).stepSize(1)
         .onChange((value:number)=>{this.rNum=value})
  }
    .height('100%')
    .width('100%')
  }
}

```
