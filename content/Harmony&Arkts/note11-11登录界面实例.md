### 线性布局容器
此类容器有主轴，元素按照主轴排列，  
column：竖向主轴  
row：横向主轴  

与主轴方向相对的轴名为交叉轴  

### 登录页面

``` ts
@Entry
@Component
struct Login_Page {
  @State message: string = 'Hello World';

  build() {
    Column({space:15}){
      Image($r('app.media.startIcon')).width(80)
      Text('Login Page').fontSize(36).fontWeight(60)
      Text('Login to use more serve').fontSize(22).fontColor(Color.Gray)
      TextInput({placeholder:'account name'}).width(250)
      TextInput({placeholder:'password'}).type(InputType.Password).width(250)

      Row({space:10}){
        Text('forgot password?').fontSize(15)
        Text('SMS login').fontSize(15)
      }.justifyContent(FlexAlign.SpaceBetween).width('65%')//左右对其

      Button('LOGIN').fontSize(22).width('30%')
      Text('Register').fontSize(22).fontColor(0x33aacc)
      LoadingProgress().width(80).color(Color.Gray)

      Row(){
        Button('T-1').fontSize(20).type(ButtonType.Circle).backgroundColor(Color.White).fontColor(Color.Gray).borderWidth(1)
        Button('T-2').fontSize(20).type(ButtonType.Circle).backgroundColor(Color.White).fontColor(Color.Gray).borderWidth(1)
        Button('T-3').fontSize(20).type(ButtonType.Circle).backgroundColor(Color.White).fontColor(Color.Gray).borderWidth(1)
      }.justifyContent(FlexAlign.SpaceBetween).width('70%')
    }
    .height('100%')
    .width('100%')
  }
}
```
