/*@state (组件内状态管理修饰器)父组件
 @Prop 子组件单向同步
 @link 子组件双向同步
 @Provide与@Consume跨层级双向同步
 */

## 单向同步
子组件与父组件，父组件发生变化时会同步至子组件  
### 子组件
``` ts
@Component
export struct sonComponent{
  @Prop inventory:number;
  private buyNum:number = 1;
  build() {
    Row(){
      if (this.inventory>0){
        Text(`${this.inventory}`).fontSize(20)
      }else{
        Text(`0`).fontSize(20)
      }
      Button(`buy${this.buyNum}`).fontSize(20).onClick(()=>{
        if (this.inventory>0)
          this.inventory-=this.buyNum;
      })
    }.width('100%').height(60).background(0xaaccaa).borderRadius(20).justifyContent(FlexAlign.SpaceEvenly)
  }
}
```

### 父组件
``` ts
import {sonComponent} from './soncomponent'
import {linkComponent} from './linkcomponent'

@Entry
@Component
struct fatherComponent{
  @State fuInventory:number = 30;
  build(){
    Column(){
      Row(){
        Text(`${this.fuInventory}`).fontSize(22)
        Button('+1').fontSize(22).type(ButtonType.Circle).width(50)
          .onClick(()=>{this.fuInventory++;})
        Button('-1').fontSize(22).type(ButtonType.Circle).width(50)
          .onClick(()=>{if(this.fuInventory>0)this.fuInventory--;})
      }.justifyContent(FlexAlign.SpaceEvenly).width('100%')
      sonComponent({inventory:this.fuInventory})
      sonComponent({inventory:this.fuInventory,buyNum:5})
      linkComponent({inventory:this.fuInventory})
      linkComponent({inventory:this.fuInventory})
    }.width('100%').height('100%')
  }
}
```
