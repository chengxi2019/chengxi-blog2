@Entry
@Component
struct Index {
  @State message: string = 'Hello Arkts';
  //时间控件需要先定义
  private RTController:TextTimerController=new TextTimerController();
  private txtController:TextTimerController=new TextTimerController();

  build() {
    Column(){
      //button按钮样式
      Button("Click me!").width('50%').fontSize(24)
      Button("注册").width('15%').fontSize(24).type(ButtonType.Circle)
      Button("登录").width('50%').fontSize(24).type(ButtonType.Normal)

      //进度条(星环加载)
      LoadingProgress().width(100).color(Color.Gray)

      //进度条()
      Progress({value:23,total:60,type:ProgressType.Eclipse}).width(100).color(Color.Gray)//月球
      Progress({value:69,total:100,type:ProgressType.Linear}).width(100).color(Color.Brown)//普通
      Progress({value:45,total:100,type:ProgressType.ScaleRing}).width(100).color(Color.Yellow)//环
      Progress({value:78,total:100,type:ProgressType.Ring}).width(100).color(Color.Yellow)//环2

      //动态时钟
      TextClock().fontSize(24)
      TextClock().format('yyyy-MM-dd HH:mm:ss').fontSize(24)

      //计时器
      Text("计时器").fontSize(24)
      TextTimer({isCountDown:false,count:0,controller:this.txtController}).fontSize(24)
      Button("start").fontSize(24).onClick(()=>{this.txtController.start();})
      Button("pause").fontSize(24).onClick(()=>{this.txtController.pause();})
      Button("reset").fontSize(24).onClick(()=>{this.txtController.reset();})

      // 倒计时器
      Text("倒计时器").fontSize(24)
      TextTimer({isCountDown:true,count:60000,controller:this.RTController}).fontSize(24)
      Button("start").fontSize(24).onClick(()=>{this.RTController.start();})
      Button("pause").fontSize(24).onClick(()=>{this.RTController.pause();})
      Button("reset").fontSize(24).onClick(()=>{this.RTController.reset();})       
    }
    .height('100%')
    .width('100%')
  }
}
