``` sql
delimiter $$ 
CREATE PROCEDURE get_student(IN stu_id CHAR(4))
BEGIN 
	SELECT student_id,student_name,major
	FROM student
	WHERE student_id = stu_id;
END $$
delimiter 

CALL get_student('S002');


delimiter $$
CREATE PROCEDURE get_count(OUT p_count INT)
BEGIN
	SELECT COUNT(*) INTO p_count FROM student;
END $$
delimiter ;

CALL get_count(@count)

SELECT @count
```
