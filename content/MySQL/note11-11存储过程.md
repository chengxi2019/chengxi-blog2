``` sql
delimiter $$ 
CREATE PROCEDURE get_student(IN stu_id CHAR(4))
BEGIN 
	SELECT student_id,student_name,major
	FROM student
	WHERE student_id = stu_id;
END $$
delimiter 

CALL get_student('S002');


delimiter $$
CREATE PROCEDURE get_count(OUT p_count INT)
BEGIN
	SELECT COUNT(*) INTO p_count FROM student;
END $$
delimiter ;

CALL get_count(@count)

SELECT @count
```

``` sql
-- 创建
DELIMITER //
CREATE PROCEDURE find_id (OUT p_out CHAR)
BEGIN 
	SELECT student_id FROM student;
END
//
DELIMITER;
-- 提前设置参数
SET @p_out = 0;
-- 使用存储过程
CALL find_id(@p_out)
-- 展示已有的储存过程
SHOW PROCEDURE STATUS;
-- 删除指定存储过程
DROP PROCEDURE find_id
```
