``` sql
-- 创建
DELIMITER //
CREATE PROCEDURE find_id (OUT p_out CHAR)
BEGIN 
	SELECT student_id FROM student;
END
//
DELIMITER;
-- 提前设置参数
SET @p_out = 0;
-- 使用存储过程
CALL find_id(@p_out)
-- 展示已有的储存过程
SHOW PROCEDURE STATUS;
-- 删除指定存储过程
DROP PROCEDURE find_id
```



``` sql
delimiter //
CREATE FUNCTION add5(score DECIMAL(5,1))
RETURNS INT
DETERMINISTIC NO SQL
BEGIN 
	RETURN score + 5;
END 
//
delimiter 
--使用函数（单独使用时无需再使用FROM）
SELECT add5(472)

SELECT student_id,entrance_score,add5(entrance_score) as outcome FROM student
WHERE student_id IN ('S001','S002','S003')
ORDER BY student_id ASC

SHOW FUNCTION STATUS;

SHOW CREATE FUNCTION add5;

DROP FUNCTION add5
```
